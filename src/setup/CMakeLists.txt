# XPLogin10-Setup.exe - the single file a user double-clicks.
#
# The whole product is embedded as RCDATA resources. payload.rc is generated
# from the real build outputs at configure time, so the setup physically cannot
# ship a stale DLL: if XPLoginProvider.dll is rebuilt, the resource changes with
# it and the setup relinks.

# One .rc per configuration, not one shared file. The paths inside it come from
# $<TARGET_FILE:...>, which resolves to bin/Debug/... or bin/Release/... - so on
# a multi-config generator (Visual Studio, Ninja Multi-Config) a single output
# path would be four different files fighting over one name, and CMake refuses
# to generate at all: "Evaluation file to be written multiple times with
# different content".
set(XPLOGIN_PAYLOAD_RC ${CMAKE_CURRENT_BINARY_DIR}/payload_$<CONFIG>.rc)

# The baked artwork is only embedded when the build produced it. A setup built
# without logonui.exe present still installs and runs; it just draws the screen
# from primitives instead of Microsoft's bitmaps.
if(XPLOGIN_HAVE_ASSETS)
    set(XPLOGIN_PAYLOAD_ASSETS "107 RCDATA \"${XPLOGIN_ASSET_PACK}\"")
else()
    set(XPLOGIN_PAYLOAD_ASSETS "// no baked artwork in this build")
endif()

file(GENERATE OUTPUT ${XPLOGIN_PAYLOAD_RC} CONTENT
"// Generated by CMake - do not edit.
//
// The ids below are the contract with PayloadManifest() in
// src/core/src/DeploymentPlan.cpp, which is the one list the setup embeds
// from, extracts from and uninstalls from. Adding a file means adding it in
// both places; test_deployment_plan.cpp checks the manifest half for
// duplicate ids and names.
101 RCDATA \"$<TARGET_FILE:XPLoginProvider>\"
102 RCDATA \"$<TARGET_FILE:XPLoginWatchdog>\"
103 RCDATA \"$<TARGET_FILE:xplogin-install>\"
105 RCDATA \"${CMAKE_SOURCE_DIR}/resources/XPLogin.ini\"
106 RCDATA \"${CMAKE_SOURCE_DIR}/resources/XPLogin.theme.ini\"
${XPLOGIN_PAYLOAD_ASSETS}
${XPLOGIN_PAYLOAD_SOUNDS}")

add_executable(XPLogin10-Setup
    main.cpp
    Payload.cpp
    setup.rc
    ${XPLOGIN_PAYLOAD_RC}
    setup.manifest          # requireAdministrator
)

# The payload has to exist before the resource compiler reads it.
add_dependencies(XPLogin10-Setup
    XPLoginProvider
    XPLoginWatchdog
    xplogin-install
)
if(XPLOGIN_HAVE_ASSETS)
    add_dependencies(XPLogin10-Setup xplogin-assets)
endif()

target_include_directories(XPLogin10-Setup PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}
    ${CMAKE_SOURCE_DIR}/src/installer)

target_link_libraries(XPLogin10-Setup PRIVATE
    xplogin_installsupport
    xplogin_core
    advapi32
    shell32
    ole32
)

set_target_properties(XPLogin10-Setup PROPERTIES OUTPUT_NAME "XPLogin10-Setup")

if(MSVC)
    # /MANIFESTUAC:NO is coupled to setup.manifest: without it the linker adds
    # its own asInvoker trustInfo and mt.exe fails on the conflicting "level"
    # attribute. setup.manifest is the single source of the elevation request,
    # the supportedOS GUID and the UTF-8 active code page.
    target_link_options(XPLogin10-Setup PRIVATE /SUBSYSTEM:CONSOLE /MANIFESTUAC:NO)
endif()
